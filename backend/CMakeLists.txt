cmake_minimum_required(VERSION 3.20)
project(SabkaBackend LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Use Ubuntu/Linux to build this project")
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
add_executable(sabka_backend server.cpp)
target_link_libraries(sabka_backend PRIVATE PkgConfig::HTTP PkgConfig::SODIUM SQLite::SQLite3 nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(sabka_backend PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sabka_smoke smoke.cpp)
target_link_libraries(sabka_smoke PRIVATE PkgConfig::HTTP nlohmann_json::nlohmann_json)
find_package(ICU REQUIRED COMPONENTS uc)
target_link_libraries(sabka_backend PRIVATE ICU::uc)

# Native Linux shopping UI: no Node, browser, React or TypeScript needed.
add_executable(sabka_cli cli.cpp)
target_link_libraries(sabka_cli PRIVATE PkgConfig::HTTP nlohmann_json::nlohmann_json)
target_compile_options(sabka_cli PRIVATE -Wall -Wextra -Wpedantic)

add_executable(sabka_help_trigger help_trigger.cpp)
target_compile_options(sabka_help_trigger PRIVATE -Wall -Wextra -Wpedantic)

add_executable(sabka_web_smoke web_smoke.cpp)
target_link_libraries(sabka_web_smoke PRIVATE PkgConfig::HTTP nlohmann_json::nlohmann_json)
target_compile_options(sabka_web_smoke PRIVATE -Wall -Wextra -Wpedantic)

# Embedded Linux diagnostics and shared kernel/userspace protocol checks.
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
foreach(tool diagnostics device_test protocol_test)
  add_executable(sabka_${tool} ${tool}.cpp)
  target_compile_options(sabka_${tool} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_test(NAME help_protocol COMMAND sabka_protocol_test)
install(TARGETS sabka_backend sabka_cli sabka_help_trigger sabka_diagnostics sabka_device_test RUNTIME DESTINATION bin)
install(FILES catalogue.json DESTINATION share/sabka-bazaar)
install(DIRECTORY ../public/ DESTINATION share/sabka-bazaar/public)
install(FILES ../embedded/systemd/sabka-bazaar.service DESTINATION share/sabka-bazaar)

add_executable(sabka_auth_smoke auth_smoke.cpp)
target_link_libraries(sabka_auth_smoke PRIVATE PkgConfig::HTTP nlohmann_json::nlohmann_json SQLite::SQLite3)
target_compile_options(sabka_auth_smoke PRIVATE -Wall -Wextra -Wpedantic)

add_executable(sabka_voice_smoke voice_smoke.cpp)
target_link_libraries(sabka_voice_smoke PRIVATE PkgConfig::HTTP)
target_compile_options(sabka_voice_smoke PRIVATE -Wall -Wextra -Wpedantic)

add_executable(sabka_search_test search_test.cpp)
target_link_libraries(sabka_search_test PRIVATE ICU::uc nlohmann_json::nlohmann_json)
target_compile_options(sabka_search_test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME catalogue_search COMMAND sabka_search_test)
